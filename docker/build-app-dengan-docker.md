# Build App Dengan Docker

buildcli.sh

```sh
#!/bin/bash
set -euo pipefail

# Export variabel dari file .env.config
set -a
source .env.config
set +a

APP_ROOT="$PWD"

OUTDIR="${APP_ROOT}/OUTDIR/${APP_NAME}/${APP_BRANCH}/${APP_ARCH}"
mkdir -p "$OUTDIR"

LOCK_FILE="buildcli.lock"
LOG_FILE="${OUTDIR}/build.log"

# Deteksi build sedang jalan
if [ -f "$LOCK_FILE" ]; then
  LOCK_PID=$(cat "$LOCK_FILE")
  tail -f "$LOG_FILE"
  exit 1
fi

# Jalankan proses di background dan simpan PID
(
  # Di dalam subshell ini: tulis PID ke file dan trap agar lock dihapus kalau gagal
  echo $$ > "$LOCK_FILE"
  trap 'rm -f "$LOCK_FILE"' EXIT

  echo "🚧 Build mulai pada $(date)"
  docker run --rm \
    --platform "linux/${APP_ARCH}" \
    --name "${APP_NAME}--${APP_BRANCH}--${APP_ARCH}" \
    --env-file .env.config \
    --env-file .env \
    --network makuro-network \
    -v "${APP_ROOT}:/app" \
    -w /app \
    "wibuntu:${APP_ARCH}" \
    /bin/bash build.sh
) > "$LOG_FILE" 2>&1 &

BUILD_PID=$!
echo "🚀 Build berjalan di background (PID: $BUILD_PID)"

if [ -f "$LOG_FILE" ]; then
  tail -f "$LOG_FILE"
fi
```

build.sh

```sh
#!/bin/bash
set -euo pipefail

source_envs() {
  echo "✅ Loading NVM"
  source "$NVM_DIR/nvm.sh"

  echo "📦 Loading env files"
  set -a
  source .env.config
  source .env
  source gambar.sh
  set +a

  APP_ROOT="$PWD"
}

clone_repo() {
  echo "📥 Cloning repo '${APP_NAME}' branch '${APP_BRANCH}'"
  WORKDIR="${APP_ROOT}/WORKDIR/${APP_NAME}/${APP_BRANCH}/${APP_ARCH}"
  OUTDIR="${APP_ROOT}/OUTDIR/${APP_NAME}/${APP_BRANCH}/${APP_ARCH}"

  if [ ! -d "${WORKDIR}" ]; then
    echo "♻️  Reusing previous workdir"
    mkdir -p "${WORKDIR}"
  fi

  if [ ! -d "${OUTDIR}" ]; then
    echo "♻️  Reusing previous outdir"
    mkdir -p "${OUTDIR}"
  fi

  git clone --branch "${APP_BRANCH}" --single-branch "https://wibugit.wibudev.com/bip/${APP_NAME}.git" "${WORKDIR}"

  if [ -d "${OUTDIR}/node_modules" ]; then
    echo "♻️  Reusing previous node_modules"
    cp -r "${OUTDIR}/node_modules" "${WORKDIR}/"
  fi
}

prepare_project() {
  cd "${WORKDIR}"

  if [ -f "${WORKDIR}/next.config.js" ]; then
    echo "⚙️ Overriding next.config.js"
    cat <<EOF >next.config.js
module.exports = {
  output: 'standalone',
} 
EOF
  fi

  if [ -f "${WORKDIR}/next.config.mjs" ]; then
    echo "⚙️ Overriding next.config.mjs"
    cat <<EOF >next.config.mjs
export default {
  output: 'standalone',
}
EOF
  fi
}

install_dependencies() {
  echo "📦 Installing dependencies with Bun"
  bun install --frozen-lockfile
}

run_prisma() {
  if [ "${APP_IS_DB_PUSH}" = "true" ]; then
    echo "🔃 Running Prisma DB Push..."
    bunx prisma db push
  else
    echo "⏭️  Skipping Prisma DB Push"
  fi

  if [ "${APP_IS_DB_SEED}" = "true" ]; then
    echo "🌱 Running Prisma DB Seed..."
    bunx prisma db seed
  else
    echo "⏭️  Skipping Prisma DB Seed"
  fi
}

build_project() {
  echo "🏗️ Running Build"
  bun run build

  echo "📦 Copying standalone output"
  cp -r public .next/standalone/
  cp -r .next/static .next/standalone/.next/

  if [ -d "${OUTDIR}" ]; then
    echo "♻️  Reusing previous output"
    rm -rf "${OUTDIR}"
  fi

  mkdir -p "${OUTDIR}"
  cp -r "${WORKDIR}/.next/standalone/" "${OUTDIR}/"
  cp -r "${WORKDIR}/node_modules/" "${OUTDIR}/"
}

cleanup() {
  echo "🧹 Cleaning up"
  rm -rf "${WORKDIR}"
}

main() {
  source_envs
  clone_repo
  prepare_project
  install_dependencies
  run_prisma
  build_project
  cleanup
  echo "✅ Build SUCCESS!"

  printf "${GAMBAR}"
}

main "$@"
```
